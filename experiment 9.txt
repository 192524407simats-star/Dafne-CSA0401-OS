#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/ipc.h>
#include <sys/shm.h>
#include <sys/wait.h>

int main()
{
    key_t key = ftok(".", 'A');
    int shmid;
    char *shm;

    shmid = shmget(key, 1024, IPC_CREAT | 0666);

    if (shmid == -1)
    {
        perror("shmget");
        exit(1);
    }

    shm = (char *)shmat(shmid, NULL, 0);

    if (fork() == 0)
    {
        sleep(1);
        printf("Child Process Read: %s\n", shm);
        shmdt(shm);
    }
    else
    {
        strcpy(shm, "Hello shared memory");
        wait(NULL);
        shmdt(shm);
        shmctl(shmid, IPC_RMID, NULL);
    }

    return 0;
}
