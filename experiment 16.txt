#include <stdio.h>

struct Employee
{
    int id;
    char name[20];
    float salary;
};

int main()
{
    struct Employee emp[10];
    int n, i, rec;

    printf("Enter number of employees: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        printf("\nEnter details of Employee %d\n", i + 1);

        printf("Employee ID: ");
        scanf("%d", &emp[i].id);

        printf("Employee Name: ");
        scanf("%s", emp[i].name);

        printf("Salary: ");
        scanf("%f", &emp[i].salary);
    }

    printf("\nEnter record number to access (1-%d): ", n);
    scanf("%d", &rec);

    if(rec >= 1 && rec <= n)
    {
        printf("\nEmployee Details\n");
        printf("------------------------\n");
        printf("Employee ID   : %d\n", emp[rec-1].id);
        printf("Employee Name : %s\n", emp[rec-1].name);
        printf("Salary        : %.2f\n", emp[rec-1].salary);
    }
    else
    {
        printf("Invalid Record Number.\n");
    }

    return 0;
}
